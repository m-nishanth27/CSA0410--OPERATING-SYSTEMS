#include <stdio.h>

struct employee
{
    int id;
    char name[20];
    float salary;
};

int main()
{
    FILE *fp;
    struct employee e;
    int n,i;

    fp=fopen("employee.dat","wb+");

    printf("Enter number of employees: ");
    scanf("%d",&n);

    for(i=0;i<n;i++)
    {
        printf("\nEmployee %d\n",i+1);

        printf("ID: ");
        scanf("%d",&e.id);

        printf("Name: ");
        scanf("%s",e.name);

        printf("Salary: ");
        scanf("%f",&e.salary);

        fwrite(&e,sizeof(e),1,fp);
    }

    rewind(fp);

    printf("\nEmployee Records\n");

    while(fread(&e,sizeof(e),1,fp))
    {
        printf("\nID: %d",e.id);
        printf("\nName: %s",e.name);
        printf("\nSalary: %.2f\n",e.salary);
    }

    fclose(fp);

    return 0;
}
