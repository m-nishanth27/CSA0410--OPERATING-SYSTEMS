#include <stdio.h>
#include <stdlib.h>

int main() {
    int request[20], n, head, disk, i, j, temp;
    int total = 0;

    printf("Enter number of requests: ");
    scanf("%d", &n);

    printf("Enter request queue: ");
    for(i = 0; i < n; i++)
        scanf("%d", &request[i]);

    printf("Enter initial head position: ");
    scanf("%d", &head);

    printf("Enter disk size: ");
    scanf("%d", &disk);

    for(i = 0; i < n - 1; i++)
        for(j = i + 1; j < n; j++)
            if(request[i] > request[j]) {
                temp = request[i];
                request[i] = request[j];
                request[j] = temp;
            }

    printf("Seek sequence: %d ", head);

    for(i = 0; i < n; i++) {
        if(request[i] < head) {
            total += abs(head - request[i]);
            head = request[i];
            printf("-> %d ", head);
        }
    }

    total += head;
    head = 0;
    printf("-> 0 ");

    for(i = 0; i < n; i++) {
        if(request[i] > head) {
            total += abs(head - request[i]);
            head = request[i];
            printf("-> %d ", head);
        }
    }

    printf("\nTotal head movement = %d\n", total);

    return 0;
}
