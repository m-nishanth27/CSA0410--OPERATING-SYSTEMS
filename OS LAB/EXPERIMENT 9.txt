#include <stdio.h>
#include <sys/ipc.h>
#include <sys/shm.h>
#include <string.h>

int main()
{
    key_t key = 1234;
    int shmid;
    char *str;

    shmid = shmget(key, 1024, 0666 | IPC_CREAT);

    str = (char *)shmat(shmid, NULL, 0);

    printf("Enter message: ");
    fgets(str, 1024, stdin);

    printf("Message written into shared memory: %s", str);

    shmdt(str);

    return 0;
}
