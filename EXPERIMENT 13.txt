#include <stdio.h>

int main()
{
    int blockSize[5], processSize[5];
    int allocation[5];
    int i,j;

    printf("Enter block sizes:\n");
    for(i=0;i<5;i++)
        scanf("%d",&blockSize[i]);

    printf("Enter process sizes:\n");
    for(i=0;i<5;i++)
        scanf("%d",&processSize[i]);

    for(i=0;i<5;i++)
        allocation[i]=-1;

    for(i=0;i<5;i++)
    {
        for(j=0;j<5;j++)
        {
            if(blockSize[j]>=processSize[i])
            {
                allocation[i]=j;
                blockSize[j]-=processSize[i];
                break;
            }
        }
    }

    printf("\nProcess\tBlock\n");

    for(i=0;i<5;i++)
    {
        if(allocation[i]!=-1)
            printf("%d\t%d\n",i+1,allocation[i]+1);
        else
            printf("%d\tNot Allocated\n",i+1);
    }

    return 0;
}
